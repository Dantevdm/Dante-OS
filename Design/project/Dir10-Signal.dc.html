// Dante OS kernel
extern "C" fn kmain() -> ! {
    serial::init();
    log!("Dante kernel starting");

    let cores = CPUS.response()
        .map(|r| r.cpus().len());
    log!("cpu: {cores} cores online");

    boot_screen(&mut screen);
    halt();
}
